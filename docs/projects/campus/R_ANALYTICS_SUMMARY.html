R언어 분석 · 코드와 실행 흐름

사용자 입력 CSV 검증·분류·집계·Shiny·HTML 리포트

실제 프로젝트 코드를 기준으로 기능·함수·실행 흐름을 따로 정리했습니다. 전체 코드 영역은 펼쳐서 읽고 복사할 수 있습니다.

1. 분석 목적과 범위

사용자가 직접 입력하거나 CSV로 올린 질문을 검증하고, 질문 의도·요일 언급·시간대·제출 시간·학점 통계를 집계하는 독립 R 프로젝트입니다. 기본 분석과 리포트는 외부 패키지 없이 실행하고, 웹 입력 화면에만 Shiny를 사용합니다.

수강 도우미의 ‘내 질문 CSV’를 수동으로 가져올 수 있습니다. 서버 대화를 자동 수집하거나 학번·성적표에 접근하지 않습니다. 샘플은 가상 데이터이며 결과에 원문 질문·개별 사용자 ID를 내보내지 않습니다.

4. 입력 형식과 해석 기준

열필수 여부·의미
message필수. 질문 원문, 최대 1,000자
user_id선택. 익명 ID 권장, 집계 사용자 수 계산
submitted_at선택. Asia/Seoul의 YYYY-MM-DD HH:MM:SS
preferred_days선택. 월·화·수·목·금 명시, 쉼표·공백·| 구분
preferred_period선택. 오전·오후·미지정, 다른 값은 기타
credits선택. 유효한 0~30 입력값을 집계

최대 5MB·50,000행입니다. 빈 질문은 제외하고 중복 열 이름은 오류로 처리합니다. 날짜 해석 실패 값은 시간 분석에서, 비정상 학점은 학점 통계에서 제외합니다.

의도는 규칙 순서에서 첫 번째로 맞는 분류를 사용합니다. 요일 집계는 선호 확정값이 아닌 언급 빈도이므로 “금요일 제외”도 금요일 언급입니다. preferred_days가 있으면 그 열을 우선합니다.

# 외부 패키지 없이 CSV 검증·집계·HTML 리포트 생성.
html_escape <- function(x) {
  x <- gsub('&', '&amp;', as.character(x), fixed=TRUE)
  x <- gsub('<', '&lt;', x, fixed=TRUE); x <- gsub('>', '&gt;', x, fixed=TRUE)
  gsub('"', '&quot;', x, fixed=TRUE)
}
validate_input <- function(d) {
  if (!is.data.frame(d) || !'message' %in% names(d)) stop('필수 열 message가 없습니다.')
  if (anyDuplicated(names(d))) stop('중복 열 이름이 있습니다.')
  if (nrow(d)>50000L) stop('최대 50,000행을 분석합니다.')
  for (key in c('user_id','submitted_at','preferred_days','preferred_period','credits'))
    if (!key %in% names(d)) d[[key]] <- rep(NA_character_, nrow(d))
  d <- d[c('user_id','message','submitted_at','preferred_days','preferred_period','credits')]
  d$message <- trimws(as.character(d$message)); valid <- !is.na(d$message) & nzchar(d$message)
  removed <- sum(!valid); d <- d[valid,,drop=FALSE]
  if (any(nchar(d$message)>1000L)) stop('질문은 1,000자 이하로 입력하세요.')
  attr(d,'removed_rows') <- removed; d
}
read_input <- function(path) {
  if (!file.exists(path) || file.info(path)$size>5*1024^2) stop('CSV는 5MB 이하로 준비하세요.')
  validate_input(read.csv(path,fileEncoding='UTF-8-BOM',stringsAsFactors=FALSE,check.names=FALSE,na.strings=c('','NA')))
}
count_values <- function(x) {
  t <- sort(table(x,useNA='no'),decreasing=TRUE)
  data.frame(category=names(t),count=as.integer(t),row.names=NULL)
}
analyze_input <- function(d) {
  removed <- attr(d,'removed_rows'); d <- validate_input(d)
  q <- gsub('[[:space:]]+','',tolower(d$message)); intent <- rep('기타',nrow(d))
  rules <- list('졸업요건'='졸업|이수요건','취소'='취소|철회','장바구니'='장바구니|찜',
    '시간충돌'='충돌|겹','좌석조회'='잔여|좌석|정원|마감','추천'='추천|공강|들을만','시간표'='시간표|신청내역|몇학점')
  for (name in names(rules)) intent[intent=='기타' & grepl(rules[[name]],q)] <- name
  explicit <- !is.na(d$preferred_days) & nzchar(trimws(d$preferred_days))
  text <- ifelse(explicit,d$preferred_days,d$message)
  day_counts <- vapply(c('월','화','수','목','금'), function(day) as.integer(sum(ifelse(explicit, grepl(paste0('(^|[|,[:space:]])',day,'(요일)?($|[|,[:space:]])'),text), grepl(paste0(day,'요일'),text)),na.rm=TRUE)),integer(1))
  period <- ifelse(!is.na(d$preferred_period)&nzchar(d$preferred_period),d$preferred_period,
    ifelse(grepl('오후',q),'오후',ifelse(grepl('오전',q),'오전','미지정')))
  period[!period %in% c('오전','오후','미지정')] <- '기타'
  dates <- suppressWarnings(as.POSIXct(d$submitted_at,format='%Y-%m-%d %H:%M:%S',tz='Asia/Seoul'))
  hours <- as.POSIXlt(dates,tz='Asia/Seoul')$hour
  credit <- suppressWarnings(as.numeric(d$credits)); valid_credit <- is.finite(credit)&credit>=0&credit<=30
  ids <- d$user_id[!is.na(d$user_id)&nzchar(d$user_id)]
  list(metrics=data.frame(metric=c('유효 질문 수','빈 질문 제외 수','식별자 제공 사용자 수','평균 질문 길이',
    '정상 날짜 수','유효 학점 수','평균 입력 학점'),value=c(nrow(d),if(is.null(removed))0 else removed,
    length(unique(ids)),if(nrow(d))round(mean(nchar(d$message)),1) else 0,sum(!is.na(dates)),
    sum(valid_credit),if(any(valid_credit))round(mean(credit[valid_credit]),2) else 0)),
    intents=count_values(intent),days=data.frame(category=names(day_counts),count=unname(day_counts)),
    periods=count_values(period),hours=count_values(sprintf('%02d시',hours[!is.na(hours)])),
    warnings=c('요일은 선호 확정값이 아닌 언급 빈도입니다. 질문에서 “금요일 제외”도 금요일 언급으로 집계됩니다.',
    '명시적 preferred_days가 있으면 해당 열을 우선합니다. 그 외에는 질문의 ‘요일’ 표현을 집계합니다.',
    '의도는 키워드 규칙 분류이며 학습 모델 정확도 수치가 아닙니다.',
    '원문 질문·개별 사용자 ID는 결과에 내보내지 않습니다. 익명 ID를 입력하세요.',
    '날짜는 Asia/Seoul의 YYYY-MM-DD HH:MM:SS 형식이며 해석 불가 날짜는 시간 분석에서 제외합니다.'))
}
render_report <- function(result,output) {
  dir.create(output,recursive=TRUE,showWarnings=FALSE)
  keys <- c('metrics','intents','days','periods','hours')
  for(name in keys)write.csv(result[[name]],file.path(output,paste0(name,'.csv')),row.names=FALSE,fileEncoding='UTF-8')
  table_html <- function(d) paste0('<table><thead><tr>',paste0('<th>',html_escape(names(d)),'</th>',collapse=''),
    '</tr></thead><tbody>',paste(vapply(seq_len(nrow(d)),function(i)paste0('<tr>',
    paste0('<td>',html_escape(unlist(d[i,],use.names=FALSE)),'</td>',collapse=''),'</tr>'),character(1)),collapse=''),'</tbody></table>')
  chart <- function(d,name) {
    svg <- chart_svg(d)
    writeLines(svg, file.path(output,paste0(name,'.svg')),useBytes=TRUE)
    svg
  }
  titles <- c(intents='질문 의도',days='요일 언급',periods='시간대 선호',hours='질문 제출 시간')
  sections <- paste(vapply(names(titles),function(k)paste0('<section><h2>',titles[k],'</h2>',
    chart(result[[k]],k),table_html(result[[k]]),'</section>'),character(1)),collapse='')
  text <- paste0('<!doctype html><html lang="ko"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">',
    '<title>사용자 입력 · R 분석 리포트</title><style>:root{color-scheme:light}body{margin:0;background:white;color:#24364c;font:16px/1.8 -apple-system,sans-serif}main{max-width:960px;margin:auto;padding:35px 22px}h2{margin-top:45px}table{border-collapse:collapse;width:100%}td,th{padding:10px;text-align:left;border-bottom:1px solid #ddd}th{background:#f4f7fa}svg{width:100%;height:auto}li{margin:8px 0}@media print{section{break-inside:avoid}}</style></head><body><main>',
    '<h1>사용자 입력 · R 분석 리포트</h1><p>생성: ',format(Sys.time(),tz='Asia/Seoul'),' (Asia/Seoul)</p><h2>입력 통계</h2>',
    table_html(result$metrics),sections,'<h2>해석할 때 주의할 점</h2><ul>',
    paste0('<li>',html_escape(result$warnings),'</li>',collapse=''),'</ul></main></body></html>')
  path <- file.path(output,'REPORT.html');writeLines(enc2utf8(text),path,useBytes=TRUE);invisible(path)
}

# Cairo·X11 설치 없이 R의 집계값으로 접근 가능한 SVG 막대그래프를 생성한다.
chart_svg <- function(d) {
  if(!nrow(d)||!any(d$count))return('<p>집계할 데이터가 없습니다.</p>')
  h <- 50+nrow(d)*36; max_count <- max(d$count)
  bars <- paste(vapply(seq_len(nrow(d)),function(i){
    y <- 20+(i-1)*36; w <- round(d$count[i]/max_count*470,1)
    paste0('<text x="10" y="',y+18,'" font-size="14">',html_escape(d$category[i]),'</text>',
      '<rect x="150" y="',y,'" width="',w,'" height="24" rx="3" fill="#587da5"/>',
      '<text x="',160+w,'" y="',y+18,'" font-size="14">',d$count[i],'</text>')
  },character(1)),collapse='')
  paste0('<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 700 ',h,'" role="img" aria-label="항목별 질문 수 막대그래프">',
    '<rect width="700" height="',h,'" fill="white"/>',bars,'</svg>')
}

server <- function(input,output,session) {
 values <- reactiveVal(read_input('data/sample.csv'));error <- reactiveVal('')
 observeEvent(input$csv,{tryCatch({values(read_input(input$csv$datapath));error('')},error=function(e)error(conditionMessage(e)))})
 observeEvent(input$add,{tryCatch({
   if(!nzchar(trimws(input$message)))stop('질문을 입력하세요.')
   row <- data.frame(user_id=NA_character_,message=input$message,submitted_at=format(Sys.time(),tz='Asia/Seoul'),
     preferred_days=NA_character_,preferred_period=input$period,credits=NA_character_)
   values(validate_input(rbind(values(),row)));error('');updateTextAreaInput(session,'message',value='')
 },error=function(e)error(conditionMessage(e)))})
 observeEvent(input$reset,{values(values()[FALSE,,drop=FALSE]);error('')})
 result <- reactive(analyze_input(values()))
 output$error <- renderText(error());output$metrics <- renderTable(result()$metrics)
 output$intents <- renderTable(result()$intents);output$days <- renderTable(result()$days)
 output$periods <- renderTable(result()$periods);output$notes <- renderText(paste(result()$warnings,collapse='\n'))
 output$intent_plot <- renderUI(HTML(chart_svg(result()$intents)));output$day_plot <- renderUI(HTML(chart_svg(result()$days)))
 output$csv_result <- downloadHandler(filename=function()'intents.csv',content=function(file)
  write.csv(result()$intents,file,row.names=FALSE,fileEncoding='UTF-8'))
 output$html <- downloadHandler(filename=function()'REPORT.html',content=function(file){
  dir <- tempfile('r-report-');on.exit(unlink(dir,recursive=TRUE));report <- render_report(result(),dir);file.copy(report,file)})
}
shinyApp(ui,server)
