def features(a, b):
    # 두 문자열의 유사도를 0부터 1 사이의 숫자로 계산
    def similarity(x, y):
        # 문자열의 일치 구간을 비교해 0~1 유사도 계산
        return SequenceMatcher(None, x, y, autojunk=False).ratio()

    # 한글을 초성·중성·종성으로 분해해 작은 글자 차이까지 비교
    jamo = [unicodedata.normalize("NFD", x) for x in (a, b)]
    return [
        similarity(a, b),
        similarity(*jamo),
        min(len(a), len(b)) / max(len(a), len(b), 1),
        float(a[:2] == b[:2]),
        float(a[-2:] == b[-2:]),
        float(markers(a) == markers(b)),
    ]

def markers(s):
    # 정규식과 일치하는 숫자·영어 단계 표기들을 목록으로 추출
    return re.findall(r"\d+|(?<=english)i+$", s)

학습한 첫 숫자는 편향, 나머지 여섯 숫자는 특징별 가중치로 파일에 저장합니다.

    # 이미지 또는 숫자 목록을 배열로 변환

def complete_name(name, track):
    lookup = catalog(track)
    typed = key(name)
    # 확인된 0학점 과목도 이름 비교에 포함
    known = {k: v[0] for k, v in lookup.items()}
    # 확인된 0학점 과목을 검색 목록에 추가
    known.update({"tlc5": "TLC5", "채플5": "채플5"})
    # 정규화한 이름이 목록에 정확히 있으면 표준 과목명을 바로 반환
    if typed in known:
        return known[typed]
    # 숫자 5를 영문 s로 읽은 확인된 오류를 TLC5로 보정
    if typed == "tlcs":  # 숫자 5를 영문 s로 읽은 오류를 처리
        return "TLC5"
    # 이름이 다섯 글자보다 짧으면 규칙 보정 대상에서 제외
    if len(typed) < 5:
        return name
    scores = {}
    for spelling, canonical in known.items():
        # 기존 규칙은 길이가 같은 이름끼리만 비교하므로 다른 길이는 제외
        if len(spelling) != len(typed):
            continue
        differences = [(a, b) for a, b in zip(typed, spelling) if a != b]
        # 차이가 없거나 다른 글자 중 한글 외 문자가 있으면 이 오타 규칙에서 제외
        if not differences or not all(
            "가" <= c <= "힣" for pair in differences for c in pair
        ):
            continue
        # 9글자 이상은 두 글자, 그보다 짧으면 한 글자를 넘게 다를 때 후보 제외
        if len(differences) > (2 if len(typed) >= 9 else 1):
            continue
        # 문자열의 일치 구간을 비교해 0~1 유사도 계산
        score = SequenceMatcher(
            None,
            unicodedata.normalize("NFD", typed),
            unicodedata.normalize("NFD", spelling),
        ).ratio()
        scores[canonical] = max(score, scores.get(canonical, 0))
    ranked = sorted(scores, key=scores.get, reverse=True)
    # 후보가 있고 1순위의 자모 유사도가 0.86 이상인지 확인
    if ranked and scores[ranked[0]] >= 0.86:
        # 후보가 하나이거나 1·2순위 점수 차이가 0.08 이상일 때 규칙 보정 확정
        if len(ranked) == 1 or scores[ranked[0]] - scores[ranked[1]] >= 0.08:
            return ranked[0]
    return name

def short_korean_match(typed, known):
    """네 글자 이름에서 한두 음절의 자모가 하나씩 달라진 후보를 찾는다."""
    if not re.fullmatch(r"[가-힣ㄱ-ㅎ]{4}", typed):
        return None

    candidates = {}
    for alias, canonical in known.items():
        if not re.fullmatch(r"[가-힣]{4}", alias):
            continue
        differences = [(a, b) for a, b in zip(typed, alias) if a != b]
        if not 1 <= len(differences) <= 2:
            continue

        for a, b in differences:
            # NFKD는 단독 ㄹ과 '료'의 초성 ㄹ을 같은 자모로 표현한다.
            left = unicodedata.normalize("NFKD", a)
            right = unicodedata.normalize("NFKD", b)
            if len(left) == len(right):
                close = sum(x != y for x, y in zip(left, right)) == 1
            else:
                # 받침 추가·누락 또는 초성만 남은 두 자모 음절
                close = abs(len(left) - len(right)) == 1 and (
                    left.startswith(right) or right.startswith(left)
                )
            if not close:
                break
        else:
            candidates[canonical] = min(len(differences), candidates.get(canonical, 3))

    if candidates:
        best = min(candidates.values())
        winners = [name for name, distance in candidates.items() if distance == best]
        if len(winners) == 1:
            return winners[0]
    return None
