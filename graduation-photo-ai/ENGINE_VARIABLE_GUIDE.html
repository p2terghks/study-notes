def suggest(name, profile):
    # catalog: 이름별 기준 과목 사전 / key: name을 정리한 검색 키(예: 'java')
    catalog = catalog_index(profile)
    key = normalize(name)
    if key in catalog:
        return catalog[key], True
    # near: 비슷한 후보 키 목록. 후보가 없으면 [], 있으면 최대 1개입니다.
    near = get_close_matches(key, list(catalog), n=1, cutoff=0.65)
    return (catalog[near[0]], False) if near else (None, False)

def parse_lines_with_review(lines, profile):
    """Return candidates and every nonempty unparsed line for manual review."""
    # rows: 과목 후보 목록 / rejected: 해석 실패 원문·사유 목록
    rows, rejected = [], []
    # grades: 허용 성적들을 길이순으로 정렬한 목록
    grades = sorted(PASS_GRADES | FAIL_GRADES, key=len, reverse=True)
    # g: 현재 성적 문자열 / grade_pattern: 성적들을 OR(|)로 연결한 패턴 문자열
    grade_pattern = "(?:" + "|".join(re.escape(g) for g in grades) + ")"
    # pattern: 전체 행에서 과목명·학점·성적을 추출할 정규표현식 객체
    pattern = re.compile(
        rf"^\s*(.+?)\s+(\d+(?:\.\d+)?)\s*(?:학점\s*)?"
        rf"({grade_pattern})(?:\s+(?:[0-4](?:\.\d+)?|5(?:\.0+)?))?\s*$",
        re.I,
    )
    # lines: 전체 입력 행 / line: 이번에 처리할 행 하나
    for line in lines:
        # text: 실제 인식 문자열 / confidence: 이 행의 인식 신뢰도
        text = line["text"] if isinstance(line, dict) else line
        confidence = (
            line.get("confidence", 1.0) if isinstance(line, dict) else 1.0
        )
        text = text.strip()
        if not text:
            continue
        if re.match(r"^(총학점|합계|취득학점|평균|신청학점)(?:\s|$)", text):
            continue
        # match: 일치 결과 객체. 패턴에 맞지 않으면 None입니다.
        match = pattern.fullmatch(text)
        if match:
            # name: 과목명 / credits: 아직 문자열인 학점 / grade: 성적 문자열
            name, credits, grade = match.groups()
            # A prior grade/credit pair indicates merged columns or two courses.
            if re.search(
                rf"\s\d+(?:\.\d+)?\s+{grade_pattern}(?=\s|$)", name, re.I
            ):
                rejected.append(
                    {
                        "text": text,
                        "reason": "한 줄에 여러 과목이 있어 분리해야 합니다.",
                    }
                )
                continue
        else:
            # missing_grade: 성적 없이 과목명·학점만 있는지 검사한 결과
            missing_grade = re.fullmatch(r"(.+?)\s+(\d+(?:\.\d+)?)\s*", text)
            if not missing_grade or not suggest(missing_grade[1], profile)[1]:
                rejected.append(
                    {"text": text, "reason": "과목명·학점·성적 열을 확인하세요."}
                )
                continue
            name, credits = missing_grade.groups()
            grade = ""
        # course: 찾은 기준 과목 또는 None / exact: 이름이 정확히 일치했는지
        course, exact = suggest(name, profile)
        # 한 과목 딕셔너리를 rows에 추가합니다. 학점은 float, 확인은 False로 저장합니다.
        rows.append(
            {
                "과목명": course["name"] if exact else name,
                "학점": float(credits),
                "이수구분": course["category"] if exact else "미분류",
                "성적": grade.upper(),
                "확인": False,
                "인식신뢰도": round(float(confidence), 3),
                "메모": (
                    ""
                    if exact
                    else (
                        f'과목 후보: {course["name"]}'
                        if course
                        else "교육과정 목록에 없음"
                    )
                ),
            }
        )
    # 후보 목록과 미해석 목록을 두 값의 튜플로 묶어 반환합니다.
    return rows, rejected

def audit(rows, profile, mandatory=(), advanced=False):
    # rows: 검토된 전체 내역 / profile: 적용 기준 / mandatory: 지정 의무 과목 / advanced: 심화 검사 여부
    catalog = catalog_index(profile)
    # totals: 구분별 누적 학점. 여기서 c는 이수구분 문자열입니다.
    totals = {c: 0.0 for c in CATEGORIES}
    # accepted: 인정 과목 사전 / issues: 검토 문제 / excluded: F·중복 제외 사유
    accepted = {}
    issues = []
    excluded = []
    # i: 1부터 시작하는 행 번호 / row: 이번에 검사하는 과목 딕셔너리
    for i, row in enumerate(rows, 1):
        # name: 현재 행의 과목명. 앞뒤 공백을 제거합니다.
        name = str(row.get("과목명", "") or "").strip()
        if not name:
            continue
        if row.get("확인") is not True:
            issues.append(f"{i}행 {name}: 원문 확인이 필요합니다.")
            continue
        # category: 현재 행의 이수구분(예: '전공필수')
        category = row.get("이수구분", "미분류")
        if category not in CATEGORIES or category == "미분류":
            issues.append(f"{i}행 {name}: 이수구분을 지정하세요.")
            continue
        # credits: 숫자로 바꾼 학점. 파서 함수의 문자열 credits와는 별개입니다.
        try:
            credits = float(row.get("학점"))
        except (ValueError, TypeError):
            credits = float("nan")
        if not math.isfinite(credits) or credits <= 0 or credits > 30:
            issues.append(
                f"{i}행 {name}: 유효한 학점(0 초과 30 이하)을 입력하세요."
            )
            continue
        # grade: 공백을 제거하고 대문자로 맞춘 성적 문자열
        grade = str(row.get("성적", "")).strip().upper()
        if grade in FAIL_GRADES:
            excluded.append(f"{name}: {grade} (취득학점 제외)")
            continue
        if grade not in PASS_GRADES:
            issues.append(f"{i}행 {name}: 성적을 확인하세요.")
            continue
        # key: 현재 과목명의 비교 키 / course: 그 이름으로 찾은 교육과정 정보
        key = normalize(name)
        course = catalog.get(key)
        if course and (
            credits != float(course["credits"]) or category != course["category"]
        ):
            issues.append(
                f'{i}행 {name}: 교육과정은 {course["category"]} {course["credits"]}학점입니다. 입력을 수정하거나 학과에서 인정 기준을 확인하세요.'
            )
            continue
        if not course and category.startswith("전공"):
            issues.append(
                f"{i}행 {name}: 교육과정에 없는 전공 과목입니다. 과목명·대체 인정 여부를 확인하세요."
            )
            continue
        # Conflicting repeated records require a user decision, never a silent sum.
        if key in accepted:
            # old: 이미 인정 목록에 있는 동일 과목. 현재 row와 비교합니다.
            old = accepted[key]
            if old["학점"] != credits or old["이수구분"] != category:
                issues.append(
                    f"{name}: 중복 내역의 학점 또는 구분이 다릅니다. 한 행만 남기세요."
                )
            else:
                excluded.append(f"{name}: 중복 취득은 1회만 계산")
            continue
        # 검사를 통과한 행만 accepted에 저장하고 totals[category]에 학점을 더합니다.
        accepted[key] = {
            "과목명": name,
            "학점": credits,
            "이수구분": category,
            "성적": grade,
        }
        totals[category] += credits
    totals["총학점"] = sum(totals.values())
    totals["교양합계"] = sum(
        totals[c] for c in ["기초교양", "나사렛교양", "핵심교양", "소양교양"]
    )
    totals["전공합계"] = sum(
        totals[c] for c in ["전공필수", "전공선택", "전공심화"]
    )
    # Advanced courses contribute to elective minima; no double counting in total.
    totals["전공선택"] += totals["전공심화"]
    # checks: 항목별 결과 목록 / 여기의 key: 기준 항목명 / value: 필요학점
    checks = [
        {
            "항목": key,
            "필요학점": value,
            "인정학점": totals[key],
            "부족학점": max(0, value - totals[key]),
        }
        for key, value in profile["minimums"].items()
    ]
    if advanced:
        if profile["optional_advanced"] is None:
            issues.append(
                "이 연도의 전공심화 최소학점이 제공되지 않아 확인이 필요합니다."
            )
        else:
            # value: 이 분기에서는 선택한 전공심화 최소학점입니다.
            value = profile["optional_advanced"]
            checks.append(
                {
                    "항목": "선택한 전공심화",
                    "필요학점": value,
                    "인정학점": totals["전공심화"],
                    "부족학점": max(0, value - totals["전공심화"]),
                }
            )
    # missing: 지정 의무 과목 중 미이수한 이름 목록
    missing = [name for name in mandatory if normalize(name) not in accepted]
    # uncompleted: 교육과정 전공필수 중 미이수 과목 정보 목록. 여기의 c는 과목 딕셔너리입니다.
    uncompleted = [
        c
        for key, c in catalog.items()
        if c["category"] == "전공필수" and key not in accepted
    ]
    # status: 최종 안내 문구. any 안의 c는 checks에서 꺼낸 검사 결과 딕셔너리입니다.
    status = (
        "입력 검토 필요"
        if issues
        else (
            "입력 기준 미충족"
            if missing or any(c["부족학점"] for c in checks)
            else "입력한 학점·지정 과목 기준 충족 (공식 요건 확인 대기)"
        )
    )
    # 아래 'status', 'checks' 등 따옴표 이름은 반환 딕셔너리의 키입니다.
    return {
        "status": status,
        "checks": checks,
        "missing_mandatory": missing,
        "uncompleted_catalog_required": uncompleted,
        "issues": issues,
        "excluded": excluded,
        "accepted": list(accepted.values()),
        "total": totals["총학점"],
        "pending": [
            "과목코드·수강학기·대체과목의 공식 대응표가 없어 재수강 인정은 별도 확인 필요",
            "학과의 의무 이수 과목 및 대체과목 인정 확인",
            "복수전공·부전공·교직 등 개인별 적용 기준 확인",
            "졸업시험·논문·인증·교양 필수과목 등 교육과정표 외 졸업요건 확인",
        ],
        "assumptions": [
            "전공심화 이수학점을 전공선택 최소학점에 포함합니다. 학과의 인정 방식이 다르면 이수구분을 수정하세요."
        ],
        "profile": {
            "year": profile["year"],
            "track": profile["track"],
            "page": profile["page"],
        },
    }

같은 이름이 다른 뜻일 때

  • rows: 파서에서는 새로 만든 후보 목록, audit에서는 전달받은 검토 내역입니다.
  • credits: 파서에서 추출 직후에는 문자열 '3', audit에서 변환 후에는 실수 3.0입니다.
  • key: 과목 조회에서는 'java', checks 컴프리헨션에서는 '총학점' 같은 항목명입니다. 컴프리헨션의 반복 변수는 그 내부 범위를 갖습니다.
  • c: 반복할 목록에 따라 구분 문자열·과목 딕셔너리·검사 딕셔너리를 담습니다. 약어만 보고 자료형을 단정하지 마세요.

JAVA 한 과목이 계산되는 예

  1. row = {'과목명': 'JAVA', '학점': 3, '이수구분': '전공필수', '성적': 'A+', '확인': True}
  2. name = 'JAVA', credits = 3.0, category = '전공필수', grade = 'A+'
  3. key = 'java', course = 교육과정의 JAVA 정보
  4. 모든 검사를 통과하면 accepted['java']에 인정 내역을 저장합니다.
  5. totals['전공필수']는 0.0 → 3.0, 총학점도 3.0이 됩니다.
  6. 총학점 기준 127과 비교한 checks의 부족학점은 124입니다.
  7. 입력 문제는 없어도 학점이 부족하므로 status는 '입력 기준 미충족'입니다.

변수와 딕셔너리 키는 다릅니다

return {'status': status}에서 따옴표 안의 'status'는 결과의 항목 이름(키), 오른쪽 status는 계산한 문구를 담은 변수입니다. 'pending'과 'assumptions'는 별도 지역변수 없이 반환 딕셔너리에 직접 만든 목록입니다. 또한 내부의 accepted는 딕셔너리지만 반환할 때 list(accepted.values())로 바뀌어 결과의 accepted에는 리스트가 담깁니다.
