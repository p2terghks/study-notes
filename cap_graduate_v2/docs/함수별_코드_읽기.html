def review_row(line, track):
    # 사전에서 과목명을 조회하며, 해당 키가 없으면 None을 받음
    name = line.get("과목명")
    # 과목명 셀이 따로 없으면 OCR이 읽은 전체 행에서 이름을 추출
    if name is None:
        text = line["text"].strip()
        # 열 제목이 들어 있는 행은 수강 과목이 아니므로 제외
        if "과목명" in text and "학점" in text:
            return None
        # 표를 구분하지 못한 경우 행에서 학점·성적·과목번호를 제거해 과목명 추출
        name = re.sub(
            r"\s+\d+\s+(?:[A-DF][0+-]?|P(?:ASS)?)(?:\s.*)?$", "", text, flags=re.I
        )
        name = re.sub(r"^.*?\b(?:[A-Z]+\d+|\d{5,})\s+", "", name)
    name = complete_name(name, track)
    # 표준화한 과목명을 키로 학점을 조회하며, 미등록 과목은 None
    credit = CREDITS.get(key(name))
    # 조회 결과의 첫 값(이름)은 _에 받고, 사용할 두 번째 값(영역)은 category에 저장
    _, category = catalog(track).get(
        key(name), (name, "0학점" if credit == 0 else "확인 필요")
    )
    # 학점은 문자열로 바꾸고 세 항목을 |로 연결해 검토표 한 줄을 생성
    return " | ".join([name, "" if credit is None else str(credit), category])

        def worker():
            try:
                # 이미지를 ocr 입력으로
                from ocr import recognize

                for path in files:
                    # 사진 파일을 바이트로 읽어 recognize에 전달하고 OCR 행 목록을 받음
                    lines, _ = recognize(Path(path).read_bytes())
                    # 인식된 행이 하나도 없으면 파일명을 포함한 오류를 표시
                    if not lines:
                        raise ValueError("글자를 찾지 못했어요: " + Path(path).name)
                    # 화면을 직접 수정하지 않고 결과·오류·완료 신호를 큐로 전달
                    events.put(
                        (
                            "rows",
                            [
                                row
                                for line in lines
                                if (row := review_row(line, selected))
                            ],
                        )
                    )
            # 발생한 오류를 exc에 받아 사용자에게 원인을 전달
            except Exception as exc:
                # 화면을 직접 수정하지 않고 결과·오류·완료 신호를 큐로 전달
                events.put(("error", str(exc)))
            finally:
                # 화면을 직접 수정하지 않고 결과·오류·완료 신호를 큐로 전달
                events.put(("done", None))

    def poll():
        # 큐에 처리할 사건이 남아 있는 동안 하나씩 꺼내 화면에 반영
        while not events.empty():
            # 큐에서 처리할 사건 하나를 꺼내 종류와 데이터로 나눔
            kind, value = events.get()
            # 큐에서 받은 데이터가 과목 행이면 검토표에 추가
            if kind == "rows":
                # 여러 과목 행을 줄바꿈으로 연결해 검토표의 지정 위치에 삽입
                editor.insert("end", "\n".join(value) + "\n")
            # 큐에서 받은 데이터가 오류이면 오류 대화상자를 표시
            elif kind == "error":
                # OCR 작업의 오류 내용을 대화상자로 표시
                messagebox.showerror("사진 읽기", value)
            else:
                # 다른 사진을 선택할 수 있도록 버튼 활성화
                button.configure(state="normal")
                # 진행 상태 문자열을 바꾸면 연결된 화면 문구도 갱신
                status.set("과목별 내용을 확인하세요.")
        # 150밀리초 후 poll을 다시 호출해 작업 결과를 확인
        root.after(150, poll)

    def calculate():
        try:
            # 검토표 전체 문자열을 읽고 줄별로 나눈 뒤 | 기준으로 세 항목을 분리
            rows = [
                [v.strip() for v in line.split("|")]
                for line in editor.get("1.0", "end").splitlines()
                if line.strip()
            ]
            # 입력이 없거나 과목명·학점·영역의 세 칸으로 나뉘지 않으면 계산 중단
            if not rows or any(len(row) != 3 for row in rows):
                raise ValueError("각 행을 과목명 | 학점 | 영역 형식으로 입력하세요.")
            changes = []
            filled = False

            for row in rows:
                # 전공 선택 상자에 현재 선택된 문자열을 읽음
                suggested = complete_name(row[0], track.get())
                # 보정한 이름이 입력 이름과 다르면 변경 목록에 기록하고 이름·영역 수정
                if suggested != row[0]:
                    changes.append(f"{row[0]} → {suggested}")
                    row[0] = suggested
                    # 전공 선택 상자에 현재 선택된 문자열을 읽음
                    row[2] = catalog(track.get()).get(
                        key(suggested), (suggested, row[2])
                    )[1]
                expected = CREDITS.get(key(row[0]))

                # 등록 학점은 있지만 입력값이 빈칸 또는 숫자가 아니면 목록의 학점으로 채움
                if expected is not None and not row[1].isdecimal():
                    row[1] = str(expected)
                    filled = True
                # 입력 학점이 목록의 학점과 다르면 변경 내용을 사용자에게 확인
                elif expected is not None and int(row[1]) != expected:
                    changes.append(f"{row[0]}: 입력 {row[1]}학점 → 목록 {expected}학점")
                    row[1] = str(expected)

            # 이름이나 학점의 변경 사항이 있을 때만 적용 여부를 질문
            if changes:
                # 사용자가 자동 변경 적용을 거절하면 이번 계산을 종료
                if not messagebox.askyesno(
                    "과목 정보 확인", "\n".join(changes) + "\n이대로 적용할까요?"
                ):
                    return
            # 빈 학점을 채웠거나 수정 사항이 있으면 검토표를 수정된 내용으로 다시 표시
            if filled or changes:
                # 검토표의 첫 줄 첫 글자(1.0)부터 끝(end)까지 기존 내용을 삭제
                editor.delete("1.0", "end")
                # 여러 과목 행을 줄바꿈으로 연결해 검토표의 지정 위치에 삽입
                editor.insert("1.0", "\n".join(" | ".join(row) for row in rows))
            # 전공 선택 상자에 현재 선택된 문자열을 읽음
            result = report(rows, track.get())
        # 발생한 오류를 exc에 받아 사용자에게 원인을 전달
        except ValueError as exc:
            # 입력값을 고쳐야 하는 이유를 안내 대화상자로 표시
            messagebox.showinfo("입력 확인", str(exc))
            return

        # 결과창에 내용을 쓸 수 있도록 잠시 편집 허용
        output.configure(state="normal")
        # 이전 계산 결과를 첫 글자부터 끝까지 비움
        output.delete("1.0", "end")

        # 결과를 사용자가 직접 수정하지 못하도록 읽기 전용으로 전환

def recognize(image_bytes):
    import io
    import numpy as np
    from PIL import Image, ImageOps

    # 입력은 이미 app.worker에서 read_bytes()로 읽은 사진 파일의 bytes
    # BytesIO: 바이트를 메모리 안의 파일처럼 감쌈 → Image.open: 이미지로 해석
    source = Image.open(io.BytesIO(image_bytes))
    # 사진이 2,400만 픽셀을 넘으면 메모리 사용을 제한하기 위해 처리 중단
    if source.width * source.height > 24_000_000:
        raise ValueError("이미지는 2,400만 화소 이하로 줄여주세요.")
    # 사진의 EXIF 방향 정보를 반영한 뒤 빨강·초록·파랑 3채널 형식으로 통일
    image = ImageOps.exif_transpose(source).convert("RGB")
    # 가로·세로 2600픽셀 안에 들어가도록 비율을 유지해 축소
    image.thumbnail((2600, 2600))
    # PIL 이미지 → (높이, 너비, RGB 3채널) 형태의 NumPy 숫자 배열
    pixels = np.array(image)
    # 먼저 표의 칸을 구분해 과목명만 읽는 방식을 시도
    table_lines = recognize_grid_table(pixels)

    # 표 구조를 읽었으면 그 결과를 반환하고 전체 이미지 OCR은 생략
    if table_lines is not None:
        return table_lines, []
    # 표 인식 결과가 None이면 전체 이미지에서 글자 영역을 검출하고 읽음
    # detections: [(네 꼭짓점 좌표, 인식 문자열, 신뢰도), ...]
    detections = get_reader().readtext(
        pixels, detail=1, paragraph=False, min_size=5, mag_ratio=2, canvas_size=3200
    )
    # boxes: 글자 영역 하나마다 위치·크기·문자열·신뢰도를 담는 딕셔너리 리스트
    boxes = []

    # box는 [[x1, y1], ..., [x4, y4]] 형태의 네 꼭짓점
    for box, text, confidence in detections:
        # x좌표만, y좌표만 따로 꺼내 왼쪽 위치·중심 높이·영역 높이를 계산
        xs = [p[0] for p in box]
        ys = [p[1] for p in box]
        boxes.append(
            {
                "x": float(min(xs)),
                "y": float(sum(ys) / 4),
                "height": float(max(ys) - min(ys)),
                "text": text,
                "confidence": float(confidence),
            }
        )
    # groups: 같은 줄에 속하는 글자 영역들을 묶은 리스트 안의 리스트
    # 예: [[자료구조 영역, 3 영역, A+ 영역], [다음 행의 영역들]]
    groups = []
    # y좌표가 작은 위쪽 글자부터 처리
    for item in sorted(boxes, key=lambda b: b["y"]):
        # 기존 행의 첫 영역과 높이가 가까운 첫 그룹을 찾고, 없으면 None
        # 허용 거리: 최소 7픽셀 또는 두 영역 중 작은 높이의 절반
        target = next(
            (
                g
                for g in groups
                if abs(g[0]["y"] - item["y"])
                <= max(7, min(g[0]["height"], item["height"]) * 0.5)
            ),
            None,
        )
        # 같은 행으로 묶을 그룹이 없으면 새 행 그룹 생성
        if target is None:
            groups.append([item])
        else:
            target.append(item)
    # 각 행 안에서는 x좌표 순으로 정렬해 왼쪽부터 문자열을 연결
    # 행 신뢰도는 해당 행의 글자 영역 신뢰도 중 가장 낮은 값
    lines = [
        {
            "text": " ".join(b["text"] for b in sorted(g, key=lambda b: b["x"])),
            "confidence": min(b["confidence"] for b in g),
        }
        for g in groups
    ]
    # 반환은 튜플: (행 딕셔너리 리스트, 글자 영역 딕셔너리 리스트)
    # 예: lines = [{"text": "자료구조 3 A+", "confidence": 0.95}]
    # app.worker는 lines만 사용하며 boxes는 _로 받아 사용하지 않음
    return lines, boxes

def grid_boundaries(pixels):
    """Find a fully ruled transcript table; fall back for other layouts."""
    import cv2
    import numpy as np

    # RGB 색상 이미지를 회색조 밝기 이미지로 변환
    gray = cv2.cvtColor(pixels, cv2.COLOR_RGB2GRAY)
    # 밝기 225를 기준으로 글자·선을 강조하는 이진 이미지 생성
    ink = cv2.threshold(gray, 225, 255, cv2.THRESH_BINARY_INV)[1]
    # 2차원 배열의 크기: h는 높이(행 수), w는 너비(열 수)
    h, w = gray.shape

    # 지정한 방향의 긴 선을 찾아 각 선의 중심 좌표를 반환
    # positions() 본문은 해당 함수 항목 참조

    # 반환: 회색조 이미지, 세로선 x좌표 리스트, 가로선 y좌표 리스트
    return gray, positions(True), positions(False)
positions(vertical)

vertical에 따라 가로·세로 중 한 방향의 긴 선을 mask에 남긴다. indices의 인접 좌표는 같은 선이므로 한 그룹에 넣고, 떨어진 좌표이면 새 그룹을 만든다. 그룹별 평균 좌표를 반환한다.

호출 형태

반환값 또는 실행 결과
각 선의 중심 좌표 모음. 예: [43,113,186].
    def positions(vertical):
        # vertical=True면 세로로 긴 검사 도구, False면 가로로 긴 검사 도구
        kernel = (1, max(30, h // 2)) if vertical else (max(30, w // 2), 1)
        # 가로나 세로로 긴 구조를 남겨 표 선을 검출
        mask = cv2.morphologyEx(
            ink, cv2.MORPH_OPEN, cv2.getStructuringElement(cv2.MORPH_RECT, kernel)
        )
        # 각 행 또는 열에서 검출된 선 픽셀 개수를 셈
        counts = np.count_nonzero(mask, axis=0 if vertical else 1)
        # 선 길이 조건을 만족하는 위치의 좌표를 구함
        indices = np.flatnonzero(counts > (h if vertical else w) * 0.65)
        groups = []
        for value in indices:
            # 첫 좌표이거나 이전 좌표와 떨어졌으면 새로운 표 선 그룹 생성
            if not groups or value > groups[-1][-1] + 1:
                groups.append([int(value)])
            else:
                groups[-1].append(int(value))
        return [round(sum(g) / len(g)) for g in groups]

def recognize_grid_table(pixels):
    """Read explicit cells only for the seven-column transcript layout.

    No catalog-based correction: uncertain names remain review candidates.
    """
    import cv2
    import numpy as np

    # xs의 이웃 좌표 두 개가 한 열, ys의 이웃 좌표 두 개가 한 행의 경계
    gray, xs, ys = grid_boundaries(pixels)
    # 검출한 표 경계가 부족하면 전체 이미지 인식으로 넘김
    if len(xs) < 4 or len(ys) < 3:
        return None
    reader = get_reader()

    # 지정한 행과 열의 셀을 확대해 문자열과 인식 점수를 반환
    # cell() 본문은 해당 함수 항목 참조

    # 왼쪽 외곽선이 누락된 경우 이미지 왼쪽 끝을 경계로 추가
    if xs[0] > 10:
        # 세로 경계 좌표 목록의 맨 앞에 0을 추가
        xs.insert(0, 0)
    # headers: 표준 열 이름 → 열 번호. 예: {"과목명": 4, "학점": 5}
    # 열 번호는 0부터 시작하므로 4는 다섯 번째 열
    headers = {}

    # 경계가 N개이면 그 사이의 열은 N-1개; 0번 행을 제목 행으로 읽음
    for col in range(len(xs) - 1):
        text, _ = cell(col, 0)
        field = header_field(text)
        # 열 제목을 표준 이름으로 해석했을 때만 열 번호를 기록
        if field:
            headers[field] = col
    # 과목명 칸 너비와 학점 열 위치로 예시 표의 구조인지 확인
    # 경계 개수·학점 열 위치·과목명 칸 너비가 예시 표 구조에 맞으면 열 번호 보완
    if (
        len(xs) in (8, 9)
        and headers.get("학점") == 5
        and xs[5] - xs[4] > 3 * (xs[6] - xs[5])
    ):
        # 확인한 표 구조에 맞는 열 이름과 번호를 사전에 반영
        headers.update({"과목명": 4, "학점": 5, "성적": 6, "과목번호": 3})
        # 8개 열을 가진 표이면 마지막 열을 교양영역으로 기록
        if len(xs) == 9:
            headers["교양영역"] = 7

    # 과목명 열을 찾지 못했으면 표 셀 인식을 중단하고 다른 방식으로 읽기
    if "과목명" not in headers:
        return None
    # 각 행 안에서는 x좌표 순으로 정렬해 왼쪽부터 문자열을 연결
    # 행 신뢰도는 해당 행의 글자 영역 신뢰도 중 가장 낮은 값
    lines = []

    # 0번 제목 행을 제외하고 과목명 셀만 읽음: 학점·성적 셀은 여기서 읽지 않음
    # lines: [{"과목명": "자료구조", "confidence": 0.95}, ...]
    for row in range(1, len(ys) - 1):
        name, score = cell(headers["과목명"], row)
        lines.append({"과목명": name or "이름 확인 필요", "confidence": score})
    return lines
cell(col, row, allowed=None)

col과 row는 0부터 세는 열·행 번호다. crop은 테두리를 제외한 셀 이미지다. 비어 있으면 빈 문자열과 0점을 반환한다. reader.recognize()의 결과 found에서 문자열을 이어 붙이고 가장 낮은 신뢰도를 반환한다.

반환값 또는 실행 결과

    def cell(col, row, allowed=None):
        # 배열은 [세로 범위, 가로 범위]로 자름; 테두리 선을 피하려고 3픽셀씩 안쪽 사용
        crop = gray[ys[row] + 3 : ys[row + 1] - 3, xs[col] + 3 : xs[col + 1] - 3]
        # 잘라낸 셀에 픽셀이 없으면 빈 문자열과 0점 반환
        if not crop.size:
            return "", 0.0
        # 작은 셀의 글자를 가로·세로 두 배로 확대
        crop = cv2.resize(crop, None, fx=2, fy=2, interpolation=cv2.INTER_CUBIC)
        # 셀 가장자리에 흰색 여백 12픽셀을 추가
        crop = np.pad(crop, 12, constant_values=255)
        h, w = crop.shape
        # 셀 전체를 하나의 글자 영역으로 지정해 OCR 문자열·점수를 받음
        found = reader.recognize(
            crop,
            horizontal_list=[[0, w, 0, h]],
            free_list=[],
            detail=1,
            allowlist=allowed,
            paragraph=False,
        )
        # found의 각 결과에서 v[1]은 문자열, v[2]는 신뢰도
        # 반환: (셀의 문자열, 최저 신뢰도); 결과가 없으면 신뢰도는 0
        return (
            " ".join(v[1] for v in found).strip(),
            min((float(v[2]) for v in found), default=0.0),
        )

    return {
        "과목명": "과목명",
        "교양영역": "교양영역",
        "과목영": "과목명",
        "교과목명": "과목명",
        "학점": "학점",
        "취득학점": "학점",
        "성적": "성적",
        "등급": "성적",
        "과목번호": "과목번호",
        "과목코드": "과목번호",

def match(name, track):
    known = names(track)
    # 공백과 영문 대소문자를 통일해 등록 이름 및 별칭부터 조회한다.
    typed = key(name)
    if not typed:
        return name
    if typed in known:
        return known[typed]

    # 한글 한 음절의 일부만 다를 때에도 나머지 자모가 점수에 반영된다.
    query = unicodedata.normalize("NFKD", typed)
    scores = {}
    for alias, canonical in known.items():
        candidate = unicodedata.normalize("NFKD", alias)
        # 두 자모열에서 일치한 길이의 두 배를 두 열의 전체 길이 합으로 나눈다.
        score = SequenceMatcher(None, query, candidate, autojunk=False).ratio()
        # 같은 과목의 공식 별칭들은 하나의 후보로 합친다.
        scores[canonical] = max(scores.get(canonical, 0), score)

    ranked = sorted(scores, key=scores.get, reverse=True)
    # 가장 닮은 이름도 기준에 미달하면 OCR 원문을 검토표에 남긴다.
    if not ranked or scores[ranked[0]] < MIN_SIMILARITY:
        return name
    # 서로 다른 두 과목이 비슷하게 닮았으면 하나를 임의로 고르지 않는다.
    if len(ranked) > 1 and scores[ranked[0]] - scores[ranked[1]] < MIN_MARGIN:
        return name
    return ranked[0]

def catalog(track):
    result = {}
    for group in ["멀콘", "정보", "인공", "전공선택", track, "학부공통"]:
        category = (
            "전공필수"
            if group == track
            else "학부공통" if group == "학부공통" else "전공선택"
        )
        for name in GROUPS[group]:
            result[key(name)] = (name, category)
    for category, entries in GENERAL.items():
        for entry in entries:
            names = entry.split("=")
            for alias in names:
                result[key(alias)] = (names[0], category)
    return result

예시 한 행만 있으면 전공필수 3 / 24학점과 21학점 부족을 포함한 문자열을 반환한다.
def report(rows, track):
    lookup = catalog(track)
    passed = {}
    for number, (name, credit, category) in enumerate(rows, 1):
        # 과목명이 비었거나 학점이 정수 문자열이 아니면 입력 확인 요청
        if not name.strip() or not credit.isdecimal():
            raise ValueError(f"{number}행의 과목명·학점을 확인하세요.")
        name, category = lookup.get(key(name), (name.strip(), category.strip()))
        # 0학점이 아닌 과목의 영역이 졸업요건에 없으면 분류 확인 요청
        if int(credit) and category not in TARGET:
            raise ValueError(f"{number}행 {name}의 영역을 확인하세요.")
        record = int(credit), category
        # 동일 과목의 중복 기록에서 학점 또는 영역이 다르면 충돌을 알림
        if key(name) in passed and passed[key(name)] != record:
            raise ValueError(f"{name}의 중복 행에서 학점 또는 영역이 다릅니다.")
        passed[key(name)] = record
    # 키가 중복되지 않는 사전을 생성
    totals = dict.fromkeys(TARGET, 0)

    for credit, category in passed.values():
        # 0학점 과목은 영역 합산에서 건너뛰고 양수 학점만 누적
        if credit:
            totals[category] += credit
    totals["전공선택"] += max(0, totals["전공필수"] - 24)
    totals["전공필수"] = min(24, totals["전공필수"])

    for category, minimum in TARGET.items():
        # 다른 영역의 초과 학점만 자율선택으로 이동
        if category != "자율선택":
            totals["자율선택"] += max(0, totals[category] - minimum)
            totals[category] = min(minimum, totals[category])

    lines = [f"총 취득 {sum(totals.values())}학점 / 졸업 기준 127학점"]

    for category, minimum in TARGET.items():
        missing = max(0, minimum - totals[category])
        lines.append(
            f"\n{category}  {totals[category]} / {minimum}학점  ·  "
            + (f"{missing}학점 부족" if missing else "충족")
        )
        # 부족 학점이 있을 때만 해당 영역의 미이수 후보를 검색
        if missing:
            # 키가 중복되지 않는 사전을 생성
            names = dict.fromkeys(
                n for n, c in lookup.values() if c == category and key(n) not in passed
            )
            # 추천할 미이수 후보가 한 과목 이상 있을 때만 후보 문장을 추가
            if names:
                lines.append("이수 후보: " + ", ".join(names))
    return "\n".join(lines)
